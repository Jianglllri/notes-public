```python
class Solution:
    def combinationSum(self, candidates: List[int], target: int) -> List[List[int]]:
        candidates.sort()
        res=[]
        tmp=[]
        def backtrack(num,k):
            if num==target:
                res.append(tmp.copy())## copy
                return 
            for i in range(k,len(candidates)): ##是否裁剪
                t=num+candidates[i]
                if t<=target:
                    tmp.append(candidates[i])
                    backtrack(t,i)
                    tmp.pop()
                else:
                    break
        backtrack(0,0)
        return res     
```

class Solution:
    def solveSudoku(self, board: List[List[str]]) -> None:
        self.rows=[[False]*10 for _ in range(9)]
        self.cols=[[False]*10 for _ in range(9)]
        self.boxes=[[[False]*10 for _ in range(3)]for _ in range(3)]
        self.spaces=[]
        for i in range(9):
            for j in range(9):
                if board[i][j]!='.':
                    num=int(board[i][j])
                    self.rows[i][num]=True
                    self.cols[j][num]=True
                    self.boxes[i//3][j//3][num]=True
                else:
                    self.spaces.append((i,j))
        self.backtrack(board,0)
    def backtrack(self,board,idx):
        if idx==len(self.spaces):
            return True
        i,j=self.spaces[idx]
        candidates=[]
        for k in '123456789':
            num=int(k)
            if not (self.rows[i][num] or self.cols[j][num] or self.boxes[i//3][j//3][num]):
                candidates.append(k)
        for k in candidates:
            num=int(k)
            self.rows[i][num]=self.cols[j][num]=self.boxes[i//3][j//3][num] = True
            board[i][j] = k
            
            if self.backtrack(board, idx + 1):
                return True
            
            # 回溯操作
            self.rows[i][num] = self.cols[j][num] = self.boxes[i//3][j//3][num] = False
            board[i][j] = '.'
        
        return False
